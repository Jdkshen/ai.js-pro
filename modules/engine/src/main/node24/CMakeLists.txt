cmake_minimum_required(VERSION 3.18)
project(autojs_node24_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Use the Android NDK arm64-v8a toolchain")
endif()
if(NOT EXISTS "${NODE24_LIBRARY}" OR NOT EXISTS "${NODE24_CONFIG_GYPI}" OR NOT EXISTS "${NODE24_SOURCE_DIR}/src/node.h")
  message(FATAL_ERROR "Provide matching Node24 library, build config and source headers")
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/read_node_build_flags.py" "${NODE24_CONFIG_GYPI}"
  RESULT_VARIABLE FLAGS_STATUS OUTPUT_VARIABLE V8_DEFINES OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_VARIABLE FLAGS_ERROR)
if(NOT FLAGS_STATUS EQUAL 0)
  message(FATAL_ERROR "Cannot derive pinned V8 ABI flags: ${FLAGS_ERROR}")
endif()
add_library(node24 SHARED IMPORTED)
set_target_properties(node24 PROPERTIES IMPORTED_LOCATION "${NODE24_LIBRARY}")
add_library(ajsnode24bridge SHARED node24_bridge.cpp)
target_compile_definitions(ajsnode24bridge PRIVATE ${V8_DEFINES})
target_include_directories(ajsnode24bridge PRIVATE "${NODE24_SOURCE_DIR}/src" "${NODE24_SOURCE_DIR}/deps/v8/include" "${NODE24_SOURCE_DIR}/deps/uv/include")
target_compile_options(ajsnode24bridge PRIVATE -fPIC -Wall -Wextra -Werror -Wno-unused-parameter -Wno-unused-private-field)
target_link_libraries(ajsnode24bridge PRIVATE node24)
target_link_options(ajsnode24bridge PRIVATE "-Wl,-z,defs" "-Wl,-z,max-page-size=16384" "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
set_property(TARGET ajsnode24bridge APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
